import {describe,expect,it} from 'vitest';
import {createSqliteSql} from '@artifactbin/sql/sqlite';
import {isQueryFailure,type DatasetColumn,type DatasetMutationPolicy,type MutationInput,type MutationResult} from '@artifactbin/contracts';

const sql=createSqliteSql();
const columns:DatasetColumn[]=[{name:'label',type:'string'},{name:'done',type:'boolean'}];
const old={label:'task',done:false};
const table={name:'tasks',columns,rows:[old]};
async function success(input:MutationInput):Promise<MutationResult>{
 const result=await sql.mutate(input);
 if(isQueryFailure(result))throw new Error(result.error);
 return result;
}
const run=(statement:string,extra:Partial<MutationInput>={})=>success({table,sql:statement,params:{},capture:'single-row',...extra});

describe('single-row mutation capture contract',()=>{
 it('captures inserted typed values after presets',async()=>{
  const policy:DatasetMutationPolicy={role:'viewer',session:{},operations:['insert'],table:{table:{schema:'main',name:'tasks'},insert_permissions:[{role:'viewer',permission:{columns:'*',set:{done:true},check:{}}}]}};
  const result=await run("insert into tasks(label,done) values ('new',false)",{policy});
  expect(result.effect).toEqual({operation:'insert',before:null,after:{label:'new',done:true}});
  expect(result.rows).toEqual([old,{label:'new',done:true}]);
 });
 it('captures typed update images without mutating input',async()=>{
  const result=await run("update tasks set label='renamed',done=true");
  expect(result.effect).toEqual({operation:'update',before:old,after:{label:'renamed',done:true}});
  expect(table.rows).toEqual([{label:'task',done:false}]);
 });
 it('retains deleted values with explicit null after',async()=>{
  const result=await run("delete from tasks where label='task'");
  expect(result.effect).toEqual({operation:'delete',before:old,after:null});
  expect(result.rows).toEqual([]);expect(result.affected).toBe(1);
 });
 it('distinguishes identical records without a primary key',async()=>{
  const duplicated={...table,rows:[{label:'same',done:false},{label:'same',done:false}]};
  // Private rowid selects one identical record for this engine test only; it is not public identity.
  const result=await run('update tasks set done=true where rowid=2',{table:duplicated});
  expect(result.effect).toEqual({operation:'update',before:{label:'same',done:false},after:{label:'same',done:true}});
  expect(result.rows).toEqual([{label:'same',done:false},{label:'same',done:true}]);
 });
 it('refuses multirow capture while retaining ordinary bulk writes',async()=>{
  const input={table:{...table,rows:[old,{...old}]},sql:'update tasks set done=true',params:{}} satisfies MutationInput;
  const refused=await sql.mutate({...input,capture:'single-row'});
  expect(refused).toMatchObject({code:'row_not_unique'});
  expect(refused).toHaveProperty('error');expect(refused).not.toHaveProperty('rows');expect(refused).not.toHaveProperty('effect');
  const ordinary=await success(input);expect(ordinary.affected).toBe(2);expect(ordinary).not.toHaveProperty('effect');
 });
 it('distinguishes zero effects, same-value updates and exact cardinality guards',async()=>{
  const zero=await run("update tasks set done=true where label='absent'");
  expect(zero.affected).toBe(0);expect(zero).toHaveProperty('effect',null);
  const unchanged=await run('update tasks set done=done');
  expect(unchanged.effect).toEqual({operation:'update',before:old,after:old});expect(unchanged.affected).toBe(1);
  const guarded=await sql.mutate({table,sql:"delete from tasks where label='absent'",params:{},capture:'single-row',expectedAffected:1});
  expect(guarded).toMatchObject({code:'row_changed'});expect(guarded).not.toHaveProperty('effect');
 });
});
